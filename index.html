startButton.addEventListener('click', async () => { startButton.style.display = 'none'; try { // ROZWIĄZANIE 1: Wyłączenie filtrów systemowych na telefonach/tabletach const stream = await navigator.mediaDevices.getUserMedia({ audio: { echoCancellation: false, noiseSuppression: false, autoGainControl: false }, video: false }); const audioCtx = new (window.AudioContext || window.webkitAudioContext)(); const source = audioCtx.createMediaStreamSource(stream); // ROZWIĄZANIE 2: Dodatkowe programowe wzmocnienie sygnału (GainNode) // Telefony mają bardzo ciche wejścia mikrofonowe w przeglądarkach. const gainNode = audioCtx.createGain(); gainNode.gain.value = 2.5; // Zwiększ tę wartość (np. do 3.0 - 4.0), jeśli słupki nadal będą za niskie const analyser = audioCtx.createAnalyser(); analyser.smoothingTimeConstant = 0.60; analyser.fftSize = 256; const bufferLength = analyser.frequencyBinCount; const dataArray = new Uint8Array(bufferLength); const numberOfBars = 20; const currentHeights = new Float32Array(numberOfBars); const peakHeights = new Float32Array(numberOfBars); const peakHoldTimers = new Float32Array(numberOfBars); // Łączenie węzłów audio: Źródło -> Wzmocnienie -> Analizer source.connect(gainNode); gainNode.connect(analyser); let lastTime = performance.now(); function draw(currentTime) { requestAnimationFrame(draw); let deltaTime = (currentTime - lastTime) / 1000; if (deltaTime > 0.1) deltaTime = 0.1; lastTime = currentTime; analyser.getByteFrequencyData(dataArray); ctx.clearRect(0, 0, window.innerWidth, window.innerHeight); const barWidth = window.innerWidth / numberOfBars; // ROZWIĄZANIE 3: Ignorujemy wycięte przez telefon wysokie częstotliwości // Zamiast 0.7 bierzemy 0.55 pasma, bo góra pasma na mobile i tak jest głucha const usableBufferLength = Math.floor(bufferLength * 0.55); const sampleSize = usableBufferLength / numberOfBars; const speedUp = 20; const speedDown = 11; // --- PARAMETRY CEGIEŁEK (EQUALIZERA LED) --- const brickHeight = 12; const brickGap = 4; const totalBrickStep = brickHeight + brickGap; const peakHoldTime = 0.35; const peakFallSpeed = 450; for (let i = 0; i < numberOfBars; i++) { let sum = 0; const startIndex = Math.floor(i * sampleSize); const endIndex = Math.floor((i + 1) * sampleSize); let count = 0; for (let j = startIndex; j < endIndex; j++) { if(dataArray[j] !== undefined) { sum += dataArray[j]; count++; } } const averageVolume = count > 0 ? sum / count : 0; // Zwiększony mnożnik wysokości ekranu z 1.3 na 1.6 dla lepszej dynamiki const targetHeight = (averageVolume / 255) * window.innerHeight * 1.6; if (targetHeight > currentHeights[i]) { currentHeights[i] += (targetHeight - currentHeights[i]) * speedUp * deltaTime; } else { currentHeights[i] += (targetHeight - currentHeights[i]) * speedDown * deltaTime; } if (currentHeights[i] > window.innerHeight) currentHeights[i] = window.innerHeight; const barHeight = currentHeights[i]; const hue = (i / numberOfBars) * 360; if (barHeight > peakHeights[i]) { peakHeights[i] = barHeight; peakHoldTimers[i] = peakHoldTime; } else { if (peakHoldTimers[i] > 0) { peakHoldTimers[i] -= deltaTime; } else { peakHeights[i] -= peakFallSpeed * deltaTime; if (peakHeights[i] < 0) peakHeights[i] = 0; } } const x = i * barWidth; ctx.fillStyle = `hsl(${hue}, 95%, 50%)`; const numberOfBricksInBar = Math.floor(barHeight / totalBrickStep); for (let b = 0; b < numberOfBricksInBar; b++) { const yPos = window.innerHeight - (b * totalBrickStep) - brickHeight; ctx.fillRect(x, yPos, barWidth - 4, brickHeight); } ctx.fillStyle = `hsl(${hue}, 100%, 65%)`; if (peakHeights[i] > 0.5) { const peakBrickIndex = Math.floor(peakHeights[i] / totalBrickStep); const peakYPos = window.innerHeight - (peakBrickIndex * totalBrickStep) - brickHeight; ctx.fillRect(x, peakYPos, barWidth - 4, brickHeight); } } } requestAnimationFrame(draw); } catch (err) { console.error('Błąd dostępu do mikrofonu:', err); alert('Nie można uruchomić wizualizatora. Wymagany jest dostęp do mikrofonu.'); } });
